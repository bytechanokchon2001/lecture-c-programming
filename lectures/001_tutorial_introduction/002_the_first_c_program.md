# The first C program

    #include <stdio.h>

    main() 
    {
        printf("Hello World\n");
    }

ฟังก์ชันการทำงานของเมธอดจะอยู่ภายใน `{ }` เสมอ โดยจากตัวอย่างเมธอด main มีเพียงการดำเนินการเดียวภายในเมธอด

การเรียกใช้งานเมธอดเราสามารถเรียกใช้งานได้ผ่าน **ชื่อเมธอด** และสามารถส่งข้อมูลเข้าไปได้ผ่านอาร์กิวเมนต์ เช่น `printf("Hello World");` เป็นเมธอดที่มาจาก `stdio.h` ซึ่งเราเรียกใช้เมธอดดังกล่าวด้วยการเรียกชื่อเมธอด `printf` และทำการส่งอาร์กิวเมนต์เข้าไปด้วยข้อมูล `"Hello World"` ซึ่งเป็นข้อมูลที่เราต้องการนำมาแสดงผลบนหน้าจอ

`"Hello World"` ถูกเรียกว่า *character string* หรือ *string constant* 

`\n` ใช้สำหรับการขึ้นบรรทัดใหม่ โดยหากเราไม่ใช้งาน `\n` เมื่อเราทำการแสดงข้อความตัวถัดไป มันจะอยู่บรรทัดเดียวกัน

    int main()
    {
        printf("Hello, world");
        printf("Welcome to C Programming");

        return 0;
    }
    // ผลลัพท์: Hello, worldWelcome to C Programming

โดย `printf` จะไม่มีฟังก์ชันสำหรับการขึ้นบรรทัดใหม่ให้อัตโนมัติ เราจำเป็นต้องใช้ `\n` เท่านั้น
นอกจากนี้ ยังมีตัวอักษรพิเศษอื่น ๆ อีก เช่น
- `\t` สำหรับ tap
- `\b` สำหรับ backspace
- `\"` สำหรับ การนำอักขระ `"` ไปใส่ไว้ใน string
- `\\` สำหรับ การนำอักขระ `\` ไปใส่ไว้ใน string
